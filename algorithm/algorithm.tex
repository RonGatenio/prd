\subsubsection{Happens-Before Graph}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

Let $\HBG = (\Nodes, \HBEdges)$ be a Happens-Before graph where $\Nodes$ is the set of the trace events and $\HBEdges$ is the set of Happens-Before edges derived from the program trace log.

We will denote Happens-Before relation with $\HBSign$.

The $\HBG$ ensures that:
$$\HB{v_1}{v_k} \iff \exists \Path = (v_1, \ldots, v_k) \in \Nodes^k \text{ where } \Path \text{ is a path in } \HBG$$

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\subsubsection{Happens-After Vector Clock}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

$\HAVC{t,i}{t'}$ will be the first node's index in thread $t'$ that happens-after the $i^{th}$ node in thread $t$.
\[
    \HAVC{t,i}{t'} = \min \left( \left\{ j \middle| \ 
    \begin{aligned}
        & \exists \Path = (v_1, \ldots, v_k) \in \Nodes^k, k \geq 1 \text{ a path in \HBG} \\
        & \text{such that:} \\
        & \begin{aligned}
            & && \loc{v_1} = (t, i), \ \loc{v_k} = (t', j)
        \end{aligned}
    \end{aligned}
    \right\} \right)
\]


%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\subsubsection{Happens-Before Vector Clock}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

$\HBVC{t,i}{t',\alpha}$ will be the last node's index in thread $t'$ of variable $\alpha$ that happens-before the $i^{th}$ node in thread $t$.
\[
    \HBVC{t,i}{t',\alpha} = \max \left( \left\{ j \middle| \ 
    \begin{aligned}
        & \exists \Path = (v_1, \ldots, v_k) \in \Nodes^k, k \geq 1 \text{ a path in \HBG} \\
        & \text{such that:} \\
        & \begin{aligned}
            & && \loc{v_1} = (t', j), \ \loc{v_k} = (t, i) \\
            % & && lbl(v_1) \in \{ R(\alpha), W(\alpha) \} \\
            & && v_1 \in \ReadNodes[\alpha] \cup \WriteNodes[\alpha] \\
        \end{aligned}
    \end{aligned}
    \right\} \right)
\]


%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\subsubsection{Persisted-Before Vector Clock}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

$\PBVC{t,i}{t',\alpha}$ will be the last node's index in thread $t'$ of variable $\alpha$ that was persisted before the $i^{th}$ node in thread $t$. \\
A persisted node is a node that happens-before a Flush event (on the same variable).
\[
    \PBVC{t,i}{t',\alpha} = \max \left( \left\{ j \middle| \ 
    \begin{aligned}
        & \exists \Path = (v_1, \ldots, v_k) \in \Nodes^k, k \geq 1 \text{ a path in \HBG} \\
        & \text{such that:} \\
        & \begin{aligned}
            & && \loc{v_1} = (t', j), \ \loc{v_k} = (t, i) \\
            % & && lbl(v_1) \in \{ R(\alpha), W(\alpha) \} \\
            & && v_1 \in \ReadNodes[\alpha] \cup \WriteNodes[\alpha] \\
            % & && \exists 1 < f \leq k. \;\; lbl(v_f) = F(\alpha)
            & && \exists 1 < f \leq k. \;\; v_f \in \FlushNodes[\alpha]
        \end{aligned}
    \end{aligned}
    \right\} \right)
\]

    \begin{aligned}
        % & r, w \in \Nodes \\
        % & lbl(r) = R(\alpha), lbl(w) = W(\alpha) \\
        & \exists \alpha \in {\Vars}. \; r \in \ReadNodes[\alpha], w \in \WriteNodes[\alpha] \\
        & \loc{w} = (t_w, i_w) \\

    \right\}
\]

\begin{algorithm}[H]
    \caption{Happens-After Vector Clocks Builder}
    \Fn{\FHAVC{$\HBG$}}{
        \BlankLine
        Set default $\havc{t,i}{t'} \gets +\infty$ \hfill $\forall t,t' \in \Threads, i \in \mathbb{N}$ \;
        \BlankLine
        \ForEach{node $v$ in $\HBG$ in post-order}{
            $(t, i) \gets \loc{v}$\;
            $\havc{t,i}{t} \gets i$\;
            \BlankLine
            \ForEach{parent node $p$ such that $(p, v) \in \HBEdges$}{
                $(t_p, i_p) \gets \loc{p}$\;
                \ForEach{$t' \in \Threads$}{
                    $\havc{t_p,i_p}{t'} \gets \min \left( \havc{t_p,i_p}{t'}, \havc{t,i}{t'} \right)$
                }
            }
        }
        \BlankLine
        \KwRet $\havc{}{}$\;
    } 
\end{algorithm}

\begin{algorithm}[H]
    \caption{Persisted-Before Vector Clocks Builder}
    \Fn{\FPBVC{$\HBG$}}{
        \BlankLine
        Set default $\hbvc{t,i}{t', \alpha} \gets -\infty$ \hfill $\forall t,t' \in \Threads, i \in \mathbb{N}, \alpha \in {\Vars}$ \;
        Set default $\pbvc{t,i}{t', \alpha} \gets -\infty$ \hfill $\forall t,t' \in \Threads, i \in \mathbb{N}, \alpha \in {\Vars}$ \;
        \BlankLine
        \ForEach{node $v$ in $\HBG$ in reversed post-order}{
            $(t, i) \gets \loc{v}$, $\alpha \gets \var{v}$\;
        
            % RW
            \BlankLine
            \Comment*[l]{Handle Read and Write nodes}
            \If{$v \in R \cup W$}{
                $\hbvc{t,i}{t,\alpha} \gets i$\;
            }
        
            % F
            \BlankLine
            \Comment*[l]{Handle Flush nodes}
            \If{$v \in F$}{
                \ForEach{$t' \in \Threads$}{
                    $\pbvc{t,i}{t',\alpha} \gets \hbvc{t,i}{t',\alpha}$\;
                }
            }
        
            % Pass to children
            \BlankLine
            \Comment*[l]{Handle children nodes}
            \ForEach{child node $c$ such that $(v, c) \in \HBEdges$}{
                $(t_c, i_c) \gets \loc{c}$\;
                \ForEach{$t' \in \Threads$}{
                    \ForEach{$\beta \in {\Vars}$}{
                        $\hbvc{t_c,i_c}{t',\beta} \gets \max \left( \hbvc{t_c,i_c}{t',\beta}, \hbvc{t,i}{t',\beta} \right)$\;
                        $\pbvc{t_c,i_c}{t',\beta} \gets \max \left( \pbvc{t_c,i_c}{t',\beta}, \pbvc{t,i}{t',\beta} \right)$\;
                    }
                }
            }
        }
        \BlankLine
        \KwRet $\pbvc{}{}$\;
    }
\end{algorithm}

\begin{algorithm}[H]
    \caption{Concurrent Persistence Race Detector}
    \Fn{\FCPRD{$\HBG$, $\PDG$}}{
        $\havc{}{} \gets \FHAVC{\HBG}$\;
        $\pbvc{}{} \gets \FPBVC{\HBG}$\;
        $\races \gets \varnothing$\;
        \BlankLine
        \ForEach{write node $v \in \WriteNodes$ in $\HBG$}{
            $(t, i) \gets \loc{v}$, $\alpha \gets \var{v}$\;

            \BlankLine
            \ForEach{read node $r \in R$ such that $\PD{v}{r}$ in $\PDG$}{
                $(t', j) \gets \loc{r}$, $\beta \gets \var{r}$\;
                \Assert{$t' \neq t$}\;
                % TODO: explain that t wasn't changed or something
                
                \BlankLine
                \Comment*[l]{If the dependent node and the read node are of the same variable, there is not bug}
                \If{$\alpha = \beta$}{
                    \Continue\;
                }
                
                \BlankLine
                \Comment*[l]{If the read node is persisted, there is not bug}
                \If{$j \leq \pbvc{t,i}{t}$}{
                    \Continue\;
                }
                
                \BlankLine
                \Comment*[l]{If the read node is persisted, there is not bug}
                \If{$j \leq \pbvc{t,i}{t}$}{
                    \Continue\;
                }

                \BlankLine
                \ForEach{write node $w \in \WriteNodes[\alpha]$}{
                    $(t_w, i_w) \gets \loc{w}$
                    \BlankLine
                    \If{$\pbvc{t,i}{t_w} < i_w < \havc{t,j}{}$}{
                        $\races \gets \races \cup {(r, w)}$\;
                    }
                }
            }
            
        
        }
        \BlankLine
        \KwRet $\races$\;
    }
\end{algorithm}
