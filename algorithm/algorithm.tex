\subsubsection{General Notations}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

The program trace contains events and relations between them. \\
Each event has a type: Read, Write, or Flush. \\
Each event operates on a variable (a memory address in PM space).

We will define some groups on the collection of data from the program trace:
\begin{itemize}
    \item Let $\Vars$ be all variables in the trace.
    \item Let $\Threads$ be all threads in the trace.
    \item Let $\Nodes$ be all event nodes in the trace.
    \item Let $\ReadNodes, \WriteNodes, \FlushNodes$ be all the Read, Write, and Flush nodes in the trace respectively.
    \item Let $\ReadNodes[\VarX], \WriteNodes[\VarX], \FlushNodes[\VarX]$ be all the Read, Write, and Flush nodes in the trace respectively such that the variable operated by the node is $\VarX$.
    \item Let $\var{\NodeV} = \VarX$ be the variable operated by node $\NodeV$. \hfill $\NodeV \in \Nodes, \VarX \in \Vars$
    \item Let $\loc{\NodeV} = (\Thread, \IndexI)$ be the location of node $\NodeV$, the $\IndexI^{th}$ node in thread $\Thread$. \hfill $\NodeV \in \Nodes, \Thread \in \Threads, \IndexI \in \N$
    \item Let $\thread{\NodeV}$ be the thread the event $\NodeV \in \Nodes$ occurred on.
    \item Let $\idx{\NodeV}$ be the index of the event $\NodeV \in \Nodes$ in the program trace.
\end{itemize}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\subsubsection{Happens-After Vector Clock}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

$\HAVC{\NodeV}{\Thread}$ will be the first node's index in thread $\Thread$ that happens-after node $\NodeV$.
$$
    \HAVC{\NodeV}{\Thread} = \min \left( \left\{ \IndexI \middle| \ 
    \begin{aligned}
        & \exists \NodeU \in \Nodes \such \thread{\NodeU} = \Thread \\
        & \exists \Path = (\NodeV, \ldots, \NodeU), \text{ path of length } k \geq 1 \\
        & \idx{\NodeU} = \IndexI
    \end{aligned}
    \right\} \right)
$$


%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\subsubsection{Happens-Before Vector Clock}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

$\HBVC{\NodeV}{\Thread,\VarX}$ will be the last node's index in thread $\Thread$ of variable $\VarX$ that happens-before the node $\NodeV$.
$$
    \HBVC{\NodeV}{\Thread,\VarX} = \max \left( \left\{ \IndexI \middle| \ 
    \begin{aligned}
        & \exists \NodeU \in \ReadNodes \cup \WriteNodes \such \thread{\NodeU} = \Thread \land \var{\NodeU} = \VarX \\
        & \exists \Path = (\NodeU, \ldots, \NodeV), \text{ a path of length } k \geq 1 \\
        & \idx{\NodeU} = \IndexI
    \end{aligned}
    \right\} \right)
$$


%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\subsubsection{Persisted-Before Vector Clock}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

$\PBVC{\NodeV}{\Thread,\VarX}$ will be the last node's index in thread $\Thread$ of variable $\VarX$ that was persisted before the node $\NodeV$. \\
A persisted node is a node that happens-before a Flush event (on the same variable).
$$
    \PBVC{\NodeV}{\Thread,\VarX} = \max \left( \left\{ \IndexI \middle| \ 
    \begin{aligned}
        & \exists \NodeU \in \ReadNodes \cup \WriteNodes \such \thread{\NodeU} = \Thread \land \var{\NodeU} = \VarX \\
        & \exists \Path = (\NodeU, \ldots, \NodeV), \text{ a path of length } k \geq 1 \\
        & \exists \NodeF \in \FlushNodes \such \var{\NodeF} = \VarX \land \Path(\IndexJ) = \NodeF, 1 < \IndexJ \leq k \\
        & \idx{\NodeU} = \IndexI
    \end{aligned}
    \right\} \right)
$$


%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\subsection{Concurrent Persistence Race}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

We say there is a Concurrent Persistence Race between read node $\NodeR$ and write node $\NodeW$ if

$$
    {\PersistenceRaces} = \left\{ (\NodeR, \NodeW) \middle| \ 
    \begin{aligned}
        & \NodeR \in \ReadNodes, \NodeW \in \WriteNodes, \var{\NodeR} = \var{\NodeW} \\
        & \exists \NodeWEx \in \WriteNodes \such \var{\NodeR} \neq \var{\NodeWEx} \land \thread{\NodeR} = \thread{\NodeWEx} \\
        & \PD{\NodeWEx}{\NodeR} \\
        & \PBVC{\NodeWEx}{\thread{\NodeW}, \var{\NodeW}} < \idx{\NodeW} < \HAVC{\NodeR}{\thread{\NodeW}}
    \end{aligned}
    \right\}
$$

\begin{algorithm}[H]
    \caption{Happens-After Vector Clocks Builder}
    \label{alg:havc}
    \Fn{\FHAVC{$\HBG$}}{
        \BlankLine
        Initialize $\havc{\NodeV}{\Thread} \gets +\infty$ \hfill $\forall \NodeV \in \Nodes, \Thread \in \Threads$ \;
        \BlankLine
        \ForEach{node $\NodeV$ in $\HBG$ in post-order}{
            \label{alg:havc:mainloop}
            $\havc{\NodeV}{\thread{\NodeV}} \gets \idx{\NodeV}$ \label{alg:havc:initialize} \;
            \BlankLine
            \ForEach{parent node $\NodeParent$ such that $(\NodeParent, \NodeV) \in \HBEdges$}{
                \label{alg:havc:parentloop}
                \ForEach{$\Thread \in \Threads$}{
                    $\havc{\NodeParent}{\Thread} \gets \min \left( \havc{\NodeParent}{\Thread}, \havc{\NodeV}{\thread{\NodeV}} \right)$ \label{alg:havc:update} \;
                }
            }
        }
        \BlankLine
        % \KwRet $\havc{\NodeV}{} \hfill \forall \NodeV \in \Nodes$\;
        \KwRet $\havc{}{}$\;
    } 
\end{algorithm}

\begin{algorithm}[H]
    \caption{Persisted-Before Vector Clocks Builder}
    \Fn{\FPBVC{$\HBG$}}{
        \BlankLine
        Initialize $\hbvc{\NodeV}{\Thread, \VarX} \gets -\infty$ \hfill $\forall \NodeV \in \Nodes, \Thread \in \Threads, \VarX \in \Vars$ \;
        Initialize $\pbvc{\NodeV}{\Thread, \VarX} \gets -\infty$ \hfill $\forall \NodeV \in \Nodes, \Thread \in \Threads, \VarX \in \Vars$ \;
        \BlankLine
        \ForEach{node $\NodeV$ in $\HBG$ in reversed post-order}{
        
            % RW
            \BlankLine
            \Comment*[l]{Handle Read and Write nodes}
            \If{$\NodeV \in \ReadNodes \cup \WriteNodes$}{
                $\hbvc{\NodeV}{\thread{\NodeV},\var{\NodeV}} \gets \idx{\NodeV}$\;
            }
        
            % F
            \BlankLine
            \Comment*[l]{Handle Flush nodes}
            \If{$\NodeV \in \FlushNodes$}{
                \ForEach{$\Thread \in \Threads$}{
                    $\pbvc{\NodeV}{\Thread,\var{\NodeV}} \gets \hbvc{\NodeV}{\Thread,\var{\NodeV}}$\;
                }
            }
        
            % Pass to children
            \BlankLine
            \Comment*[l]{Handle children nodes}
            \ForEach{child node $\NodeChild$ such that $(\NodeV, \NodeChild) \in \HBEdges$}{
                \ForEach{$\Thread \in \Threads$}{
                    \ForEach{$\VarY \in {\Vars}$}{
                        $\hbvc{\NodeChild}{\Thread,\VarY} \gets \max \left( \hbvc{\NodeChild}{\Thread,\VarY}, \hbvc{\NodeV}{\Thread,\VarY} \right)$\;
                        $\pbvc{\NodeChild}{\Thread,\VarY} \gets \max \left( \pbvc{\NodeChild}{\Thread,\VarY}, \pbvc{\NodeV}{\Thread,\VarY} \right)$\;
                    }
                }
            }
        }
        \BlankLine
        % \KwRet $\pbvc{\NodeV}{} \hfill \forall \NodeV \in \Nodes$\;
        \KwRet $\pbvc{}{}$\;
    }
\end{algorithm}

\begin{algorithm}[H]
    \caption{Concurrent Persistence Race Detector}
    % \newcommand{\DependentNodeVar}{\NodeV}
    \newcommand{\DependentNodeVar}{\NodeWEx}
    \Fn{\FCPRD{$\HBG$, $\PDG$}}{
        $\havc{}{} \gets \FHAVC{\HBG}$\;
        $\pbvc{}{} \gets \FPBVC{\HBG}$\;
        $\races \gets \EmptySet$\;
        \BlankLine
        \ForEach{write node $\DependentNodeVar \in \WriteNodes$ in $\HBG$}{
            \ForEach{read node $\NodeR \in \ReadNodes$ such that $\PD{\DependentNodeVar}{\NodeR}$ in $\PDG$}{
                % TODO: explain that t wasn't changed or something
                \Assert{$\thread{\NodeR} = \thread{\DependentNodeVar}$} \; % \Comment*[r]{The 2 nodes are on the same thread}
                
                \BlankLine
                \Comment*[l]{If the dependent node and the read node are of the same variable, there is not bug}
                \If{$\var{\DependentNodeVar} = \var{\NodeR}$}{
                    \Continue\;
                }
                
                \BlankLine
                \Comment*[l]{If the read node is persisted, there is not bug}
                \If{$\idx{\NodeR} \leq \pbvc{\DependentNodeVar}{\thread{\DependentNodeVar},\var{\NodeR}}$}{
                    \Continue\;
                }

                \BlankLine
                \ForEach{write node $\NodeW \in \WriteNodes$ such that $\var{\NodeW} = \var{\NodeR}$}{
                    \BlankLine
                    \If{$\pbvc{\DependentNodeVar}{\thread{\NodeW},\var{\NodeW}} < \idx{\NodeW} < \havc{\NodeR}{\thread{\NodeW}}$}{
                        $\races \gets \races \cup {(\NodeR, \NodeW)}$\;
                    }
                }
            }
        
        }
        \BlankLine
        \KwRet $\races$\;
    }
\end{algorithm}
