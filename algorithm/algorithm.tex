\subsubsection{General Notations}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

The program trace contains events and relations between them. \\
Each event has a type: Read, Write, or Flush. \\
Each event operates on a variable (a memory address in PM space).

We will define some groups on the collection of data from the program trace:
\begin{itemize}
    \item Let $Vars$ be all variables in the trace.
    \item Let $T$ be all threads in the trace.
    \item Let $V$ be all event nodes in the trace.
    \item Let $R, W, F$ be all the Read, Write, and Flush nodes in the trace respectively.
    \item Let $R_\alpha, W_\alpha, F_\alpha$ be all the Read, Write, and Flush nodes in the trace respectively such that the variable operated by the node is $\alpha$.
    % \item Let $Ops = \{R, W, F\}$ be the event types in the trace: Read, Write, and Flush respectively.

    \item Let $var(v \in V) = \alpha \in Vars$ be the variable operated by node $v$.
    \item Let $loc(v \in V) = (t, i) \in T \times \mathbb{N}$ be the location of node $v$, meaning that $v$ is the $i^{th}$ node in thread $t$.
    
\end{itemize}

\subsubsection{Happens-Before Graph}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

Let $HBG = (V, E)$ be a Happens-Before graph where $V$ is the set of the trace events and $E$ is the set of Happens-Before edges derived from the program trace log.

We will denote Happens-Before relation with $\prec$.

The $HBG$ ensures that:
$$v_1 \prec v_k \iff \exists P = (v_1, \ldots, v_k) \in V^k \text{ where } P \text{ is a path in } HBG$$

\subsubsection{Happens-After Vector Clock}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

${HAVC}_{t,i}(t')$ will be the first node's index in thread $t'$ that happens-after the $i^{th}$ node in thread $t$.
\[
    {HAVC}_{t,i}(t') = \min \left( \left\{ j \middle| \ 
    \begin{aligned}
        & \exists P = (v_1, \ldots, v_k) \in V^k, k \geq 1 \text{ a path in HBG} \\
        & \text{such that:} \\
        & \begin{aligned}
            & && loc(v_1) = (t, i), \ loc(v_k) = (t', j)
        \end{aligned}
    \end{aligned}
    \right\} \right)
\]


%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\subsubsection{Happens-Before Vector Clock}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

${HBVC}_{t,i}(t',\alpha)$ will be the last node's index in thread $t'$ of variable $\alpha$ that happens-before the $i^{th}$ node in thread $t$.
\[
    {HBVC}_{t,i}(t',\alpha) = \max \left( \left\{ j \middle| \ 
    \begin{aligned}
        & \exists P = (v_1, \ldots, v_k) \in V^k, k \geq 1 \text{ a path in HBG} \\
        & \text{such that:} \\
        & \begin{aligned}
            & && loc(v_1) = (t', j), \ loc(v_k) = (t, i) \\
            % & && lbl(v_1) \in \{ R(\alpha), W(\alpha) \} \\
            & && v_1 \in R_\alpha \cup W_\alpha \\
        \end{aligned}
    \end{aligned}
    \right\} \right)
\]


%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\subsubsection{Happens-Before Vector Clock}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

${PBVC}_{t,i}(t',\alpha)$ will be the last node's index in thread $t'$ of variable $\alpha$ that was persisted before the $i^{th}$ node in thread $t$. \\
A persisted node is a node that happens-before a Flush event (on the same variable).
\[
    {PBVC}_{t,i}(t',\alpha) = \max \left( \left\{ j \middle| \ 
    \begin{aligned}
        & \exists P = (v_1, \ldots, v_k) \in V^k, k \geq 1 \text{ a path in HBG} \\
        & \text{such that:} \\
        & \begin{aligned}
            & && loc(v_1) = (t', j), \ loc(v_k) = (t, i) \\
            % & && lbl(v_1) \in \{ R(\alpha), W(\alpha) \} \\
            & && v_1 \in R_\alpha \cup W_\alpha \\
            % & && \exists 1 < f \leq k. \;\; lbl(v_f) = F(\alpha)
            & && \exists 1 < f \leq k. \;\; v_f \in F_\alpha
        \end{aligned}
    \end{aligned}
    \right\} \right)
\]

\[
    {PersistenceRaces} = \left\{ (r, w) \middle| \ 
    \begin{aligned}
        % & r, w \in V \\
        % & lbl(r) = R(\alpha), lbl(w) = W(\alpha) \\
        & r \in R_\alpha, w \in W_\alpha \\
        & loc(w) = (t_w, i_w) \\

            & && v \overset{pd}{\to} r \\
            & && {PBVC}_{loc(v)}(t_w, \alpha) < i_w < {HAVC}_{loc(r)}(t_w)
        \end{aligned}

    \right\}

\begin{algorithm}[H]
    \caption{Happens-After Vector Clocks Builder}
    \Fn{\FHAVC{$HBG$}}{
        \BlankLine
        Set default ${havc}_{t,i}(t') \gets +\infty$ \hfill $\forall t,t' \in T, i \in \mathbb{N}$ \;
        \BlankLine
        \ForEach{node $v$ in HBG in post-order}{
            $(t, i) \gets loc(v)$\;
            ${havc}_{t,i}(t) \gets i$\;
            \BlankLine
            \ForEach{parent node $p$ such that $(p, v) \in E$}{
                $(t_p, i_p) \gets loc(p)$\;
                \ForEach{$t' \in T$}{
                    ${havc}_{t_p,i_p}(t') \gets \min\left({havc}_{t_p,i_p}(t'), {havc}_{t,i}(t')\right)$
                }
            }
        }
        \BlankLine
        \KwRet ${havc}$\;
    } 
\end{algorithm}

\begin{algorithm}[H]
    \caption{Persisted-Before Vector Clocks Builder}
    \Fn{\FPBVC{$HBG$}}{
        \BlankLine
        Set default ${hbvc}_{t,i}(t', \alpha) \gets -\infty$ \hfill $\forall t,t' \in T, i \in \mathbb{N}, \alpha \in {Vars}$ \;
        Set default ${pbvc}_{t,i}(t', \alpha) \gets -\infty$ \hfill $\forall t,t' \in T, i \in \mathbb{N}, \alpha \in {Vars}$ \;
        \BlankLine
        \ForEach{node $v$ in HBG in reversed post-order}{
            $(t, i) \gets loc(v)$, $\alpha \gets var(v)$\;
        
            % RW
            \BlankLine
            \Comment*[l]{Handle Read and Write nodes}
            \If{$v \in R \cup W$}{
                ${hbvc}_{t,i}(t,\alpha) \gets i$\;
            }
        
            % F
            \BlankLine
            \Comment*[l]{Handle Flush nodes}
            \If{$v \in F$}{
                \ForEach{$t' \in T$}{
                    ${pbvc}_{t,i}(t',\alpha) \gets {hbvc}_{t,i}(t',\alpha)$\;
                }
            }
        
            % Pass to children
            \BlankLine
            \Comment*[l]{Handle children nodes}
            \ForEach{child node $c$ such that $(v, c) \in E$}{
                $(t_c, i_c) \gets loc(c)$\;
                \ForEach{$t' \in T$}{
                    \ForEach{$\beta \in {Vars}$}{
                        ${hbvc}_{t_c,i_c}(t',\beta) \gets \max\left({hbvc}_{t_c,i_c}(t',\beta), {hbvc}_{t,i}(t',\beta)\right)$\;
                        ${pbvc}_{t_c,i_c}(t',\beta) \gets \max\left({pbvc}_{t_c,i_c}(t',\beta), {pbvc}_{t,i}(t',\beta)\right)$\;
                    }
                }
            }
        }
        \BlankLine
        \KwRet ${pbvc}$\;
    }
\end{algorithm}

\begin{algorithm}[H]
    \caption{Concurrent Persistence Race Detector}
    \Fn{\FCPRD{$HBG$, $PDG$}}{
        ${havc} \gets \FHAVC{HBG}$\;
        ${pbvc} \gets \FPBVC{HBG}$\;
        \BlankLine
        \ForEach{write node $v$ in HBG}{
            $(t, i) \gets loc(v)$, $\alpha \gets var(v)$\;

            \BlankLine
            \ForEach{read node $r$ in $PDG$ such that $v \overset{pd}{\to} r$}{
                $(t', j) \gets loc(r)$, $\beta \gets var(r)$\;
                \Assert{$t' \neq t$}\;
                % TODO: explain that t wasn't changed or something
                
                \BlankLine
                \Comment*[l]{If the dependent node and the read node are of the same variable, there is not bug}
                \If{$\alpha = \beta$}{
                    \Continue\;
                }
                
                \BlankLine
                \Comment*[l]{If the read node is persisted, there is not bug}
                \If{$j \leq {pbvc}_{t,i}(t)$}{
                    \Continue\;
                }
                
                \BlankLine
                \Comment*[l]{If the read node is persisted, there is not bug}
                \If{$j \leq {pbvc}_{t,i}(t)$}{
                    \Continue\;
                }

                \BlankLine
                \ForEach{write node $w \in W_\alpha$}{
                    $(t_w, i_w) \gets loc(w)$
                    \BlankLine
                    \If{${pbvc}_{t,i}(t_w) < i_w < {havc}_{t,j}$}{
                        Report bug!\;
                    }
                }
            }
            
        
        }
        \BlankLine
        % \KwRet\;
    }
\end{algorithm}
